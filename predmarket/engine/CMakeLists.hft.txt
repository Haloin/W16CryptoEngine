cmake_minimum_required(VERSION 3.16)
project(hft_engine VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(HFT_FLAGS
    -O3
    -march=native
    -mtune=native
    -fno-exceptions
    -fno-rtti
    -fno-stack-protector
    -fno-asynchronous-unwind-tables
    -fomit-frame-pointer
    -flto
    -DNDEBUG
)

add_compile_options(${HFT_FLAGS})
add_link_options(-flto)

add_executable(hft_engine
    src/hft_main.cpp
)

target_include_directories(hft_engine PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/include
)

target_link_libraries(hft_engine PRIVATE
    pthread
    m
)

set_target_properties(hft_engine PROPERTIES
    INTERPROCEDURAL_OPTIMIZATION TRUE
)
